map()

transform every item in an array

const values = [1, 2, 3];

const doubled = values.map(
  n => n * 2
);

console.log(doubled);

OUTPUT

> [2, 4, 6]

filter()

keep items that match a condition

const values = [1, 2, 3, 4];

const even = values.filter(
  n => n % 2 === 0
);

console.log(even);

OUTPUT

> [2, 4]

reduce()

combine many items into one value

const values = [1, 2, 3, 4];

const sum = values.reduce(
  (total, n) => total + n, 0
);

console.log(sum);

OUTPUT

> 10

find()

find the first item that matches

const values = [5, 10, 15, 20];

const first = values.find(
  n => n > 8
);

console.log(first);

OUTPUT

> 10

flat()

flatten nested arrays into one array

const values = [
  1,
  [2, 3],
  [4, 5]
];
const flat = values.flat();
console.log(flat);

OUTPUT

> [1, 2, 3, 4, 5]

Easy way
to remember

JavaScript array methods

  • map()→Change items
  • filter()→Select items
  • reduce()→Combine items
  • find()→Find one item
  • flat()→Flatten arrays
